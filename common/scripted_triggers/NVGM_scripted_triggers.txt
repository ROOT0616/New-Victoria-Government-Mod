nvgm_money_transfer_from_to = {
	any_scope_treaty = {
		binds = $TARGET$
		is_fulfilled_by = $SOURCE$
		any_scope_article = {
			has_type = money_transfer
			source_country = $SOURCE$
			target_country = $TARGET$
			input_quantity > 0
			source_country = {
				in_default = no
			}
		}
	}
}

nvgm_has_funded_overlord = {
	is_subject = yes
	OR = {
		nvgm_money_transfer_from_to = {
			SOURCE = root.overlord
			TARGET = root
		}
		overlord = {
			has_diplomatic_pact = {
				who = root
				type = NVGM_decrease_payments
			}
		}
	}
	overlord = {
		NOT = {
			has_war_with = root
		}
	}
}

nvgm_investment_ranking_eligible = {
	OR = {
		this = root
		is_subject_of = root
		any_scope_treaty = {
			binds = root
			any_scope_article = {
				has_type = foreign_investment_rights
				source_country = root
			}
		}
	}
}
