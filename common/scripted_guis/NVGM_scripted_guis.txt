nvgm_subsidized_food_industry_scriptedgui = {
	scope = country
	saved_scopes = {
		nvgm_actor
	}
	is_shown = {
		always = yes
	}
	ai_is_valid = {
		always = no
	}
	is_valid = {
		exists = scope:nvgm_actor
		is_direct_subject_of = scope:nvgm_actor
		NOT = {
			has_war_with = scope:nvgm_actor
		}
		always = yes
		any_scope_building = {
			is_building_type = building_food_industry
			is_subsidized = no
		}
		nvgm_contract_schema_supported = yes
		scope:nvgm_actor = {
			nvgm_contract_schema_supported = yes
		}
		nvgm_subsidy_generation_available = yes
	}
	effect = {
		if = {
			limit = {
				nvgm_contract_schema_supported = yes
				nvgm_subsidy_generation_available = yes
			}
			if = {
				limit = {
					exists = scope:nvgm_actor
					is_direct_subject_of = scope:nvgm_actor
					NOT = {
						has_war_with = scope:nvgm_actor
					}
					always = yes
				}
				if = {
					limit = {
						any_scope_building = {
							is_building_type = building_food_industry
							is_subsidized = no
						}
						nvgm_has_funded_overlord = yes
					}
					custom_tooltip = "nvgm_subsidized_scriptedgui_tooltip"
					nvgm_subsidy_begin_batch_effect = {
						SOURCE = scope:nvgm_actor
					}
					every_scope_building = {
						limit = {
							is_building_type = building_food_industry
							is_subsidized = no
						}
						nvgm_start_managed_subsidy_effect = {
							SOURCE = scope:nvgm_actor
							FUNDED = yes
						}
					}
				}
				else_if = {
					limit = {
						any_scope_building = {
							is_building_type = building_food_industry
							is_subsidized = no
						}
						liberty_desire <= 90
					}
					custom_tooltip = "nvgm_subsidized_scriptedgui_tooltip"
					add_liberty_desire = 10
					nvgm_subsidy_begin_batch_effect = {
						SOURCE = scope:nvgm_actor
					}
					every_scope_building = {
						limit = {
							is_building_type = building_food_industry
							is_subsidized = no
						}
						nvgm_start_managed_subsidy_effect = {
							SOURCE = scope:nvgm_actor
							FUNDED = no
						}
					}
				}
				else = {
					custom_tooltip = "nvgm_not_subsidized_scriptedgui_tooltip"
				}
			}
		}
	}
}
