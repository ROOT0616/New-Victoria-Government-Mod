nvgm_end_migration_effect = {
	var:nvgm_migration_destination ?= {
		if = {
			limit = {
				owner = prev
			}
			remove_variable = nvgm_target_state_variable
		}
	}
	remove_variable = nvgm_migration_destination
	remove_variable = nvgm_migration_pending
	remove_variable = nvgm_migration_events_now_var
	remove_variable = nvgm_abandonment_policy_var
	every_scope_state = {
		limit = {
			has_variable = nvgm_target_state_variable
		}
		remove_variable = nvgm_target_state_variable
	}
}

nvgm_migration_cleanup_effect = {
	every_scope_state = {
		limit = {
			has_variable = nvgm_target_state_variable
		}
		if = {
			limit = {
				OR = {
					NOT = {
						root = {
							has_journal_entry = je_nvgm_abandonment_policy
						}
					}
					NOT = {
						this = root.var:nvgm_migration_destination
					}
				}
			}
			remove_variable = nvgm_target_state_variable
		}
	}
	if = {
		limit = {
			has_variable = nvgm_migration_destination
			NOT = {
				has_journal_entry = je_nvgm_abandonment_policy
			}
		}
		nvgm_end_migration_effect = yes
	}
}
